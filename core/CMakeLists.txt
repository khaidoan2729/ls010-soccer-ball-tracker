cmake_minimum_required(VERSION 3.16)
project(ballcore CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")

find_package(Threads REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio)
find_package(nlohmann_json REQUIRED)
# Raspberry Pi OS (Debian 13) ships cpp-httplib as a compiled library with a pkg-config file but
# no CMake config. Try, in order: CMake config, pkg-config (carries the package's compile flags),
# then just the header.
find_package(httplib CONFIG QUIET)
if(NOT TARGET httplib::httplib)
  find_package(PkgConfig QUIET)
  if(PKG_CONFIG_FOUND)
    pkg_check_modules(HTTPLIB QUIET IMPORTED_TARGET GLOBAL cpp-httplib)
    if(HTTPLIB_FOUND)
      add_library(httplib::httplib ALIAS PkgConfig::HTTPLIB)
      message(STATUS "cpp-httplib ${HTTPLIB_VERSION} via pkg-config")
    endif()
  endif()
endif()
if(NOT TARGET httplib::httplib)
  find_path(HTTPLIB_INCLUDE_DIR httplib.h REQUIRED)
  find_library(HTTPLIB_LIBRARY NAMES cpp-httplib)
  add_library(httplib::httplib INTERFACE IMPORTED)
  target_include_directories(httplib::httplib INTERFACE ${HTTPLIB_INCLUDE_DIR})
  if(HTTPLIB_LIBRARY)
    target_link_libraries(httplib::httplib INTERFACE ${HTTPLIB_LIBRARY})
  endif()
  message(STATUS "cpp-httplib: header ${HTTPLIB_INCLUDE_DIR}, library ${HTTPLIB_LIBRARY}")
endif()

add_executable(ballcore src/main.cpp src/server/http_server.cpp)
target_include_directories(ballcore PRIVATE src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ballcore PRIVATE
  ${OpenCV_LIBS} nlohmann_json::nlohmann_json httplib::httplib Threads::Threads)
target_compile_options(ballcore PRIVATE -Wall -Wextra)
target_compile_definitions(ballcore PRIVATE BALLCORE_WEB_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../UI")
